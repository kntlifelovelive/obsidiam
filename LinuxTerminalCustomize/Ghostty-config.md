
##### ghostty config 

```
mkdir -p ~/.config/ghostty

touch ~/.config/ghostty/config

```

##### ghostty customized config plain text code 

```

# ===== Title ======
title = "BabbyBuBu"

# === Appearance ===
theme = "Aura"
background-opacity = 0.8
background-blur = 20
cursor-color = #769ff0

# === font family === 
font-size = 13
font-thicken = true


# ==== keybind ====
keybind = ctrl+z=close_surface
keybind = ctrl+shift+d=close_tab



# === window setting ===
copy-on-select = true
window-padding-x = 15
window-padding-y = 15
window-width = 70
window-height = 20

```
